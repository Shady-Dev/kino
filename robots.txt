# Leffavuoro. Everything worth reading is the one page. /data is the JSON it loads,
# and Google renders the page before indexing it, so the files the client needs to
# draw a schedule must be fetchable: Search Console's live test on 2026-09-13 showed an
# empty picker and the load-failure line, with 14 of 15 resources "blocked by
# robots.txt". The run-*.log files are build output and the TMDB and price caches are
# pipeline state; nothing public reads them, so they stay closed.
User-agent: *
Allow: /
Disallow: /data/
Disallow: /run.log
Disallow: /run-

# What the client reads, by file or by stable filename family. A longer Allow wins over
# the shorter Disallow (longest match; on a tie Allow wins). Posters moved under /data/
# on 2026-08-29 and are referenced from <img> and the JSON-LD "image" field, so a crawler
# that cannot fetch them cannot validate the Movie item either.
Allow: /data/providers.json
Allow: /data/regions.json
Allow: /data/areas.json
Allow: /data/venues-
Allow: /data/area-
Allow: /data/films.json
Allow: /data/films-extra.json
Allow: /data/tmdb-genres.json
Allow: /data/posters/
